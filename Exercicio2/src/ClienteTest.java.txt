import org.junit.Assert;
import org.junit.jupiter.api.Test;

import java.time.LocalDate;

import static org.junit.jupiter.api.Assertions.assertEquals;

class ClienteTest {

    @Test
    void deveCriarClienteComNome() {
        Cliente cliente = new Cliente(
                "Joao",
                "123.456.789-09",
                LocalDate.of(2000, 5, 10)
        );

        Assert.assertEquals("Joao", cliente.getNome());
    }
    @Test
    void deveCriarClienteComCpf() {
        Cliente cliente = new Cliente(
                "Joao",
                "123.456.789-09",
                LocalDate.of(2000, 5, 10)
        );

        assertEquals(
                "123.456.789-09",
                cliente.getCpf()
        );
    }
 @Test
    void deveCriarClienteComDataNascimento() {
        LocalDate data = LocalDate.of(2000, 5, 10);

        Cliente cliente = new Cliente(
                "Joao",
                "123.456.789-09",
                data
        );

        Assert.assertEquals(data, cliente.getDataNascimento());
    }

}
